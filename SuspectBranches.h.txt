#ifndef SUSPECT_BRANCHES_H
#define SUSPECT_BRANCHES_H

#include "Suspect.h"

// ============================================================================
// BRANCH 1: BUSINESS RELATIONS (Virtually inherits Suspect)
// ============================================================================
class BusinessRelations : public virtual Suspect {
protected:
    char* companyName;
public:
    BusinessRelations();
    BusinessRelations(const char* name, const char* role, char tile, int pos, WordRow row,
                      const char* mot, const char* ali, const char* w1, const char* w2,
                      const char* company);
    BusinessRelations(const BusinessRelations& other);
    BusinessRelations& operator=(const BusinessRelations& other);
    virtual ~BusinessRelations();

    virtual void        display() const override;
    virtual const char* getType() const override;
    const char*         getCompanyName() const;
    virtual int         getCredibilityRequired() const override; // Requires 50
};

#endif // SUSPECT_BRANCHES_H
