// ============================================================================
// BusinessRelations Implementation
// ============================================================================
BusinessRelations::BusinessRelations() : Suspect(), companyName(nullptr) {}

BusinessRelations::BusinessRelations(const char* name, const char* role,
                                     char tile, int pos, WordRow row,
                                     const char* mot, const char* ali,
                                     const char* w1, const char* w2,
                                     const char* company)
    : Suspect(name, role, tile, pos, row, mot, ali, w1, w2), companyName(nullptr) {
    copyStrPtr(companyName, company);
}

BusinessRelations::BusinessRelations(const BusinessRelations& other)
    : Suspect(other), companyName(nullptr) {
    copyStrPtr(companyName, other.companyName);
}

BusinessRelations& BusinessRelations::operator=(const BusinessRelations& other) {
    if (this != &other) {
        Suspect::operator=(other);
        copyStrPtr(companyName, other.companyName);
    }
    return *this;
}

BusinessRelations::~BusinessRelations() {
    delete[] companyName;
    companyName = nullptr;
}

void BusinessRelations::display() const {
    std::cout << "  Name         : " << (charName    ? charName    : "Unknown") << "\n";
    std::cout << "  Role         : " << (charRole    ? charRole    : "Unknown") << "\n";
    std::cout << "  Affiliation  : " << (companyName ? companyName : "None")    << "\n";
    std::cout << "  Motive       : " << (motive      ? motive      : "None")    << "\n";
    std::cout << "  Alibi        : " << (alibi       ? alibi       : "None")    << "\n";
}

const char* BusinessRelations::getType()                const { return "BUSINESS RELATION"; }
const char* BusinessRelations::getCompanyName()          const { return companyName; }
int         BusinessRelations::getCredibilityRequired() const { return 50; }
